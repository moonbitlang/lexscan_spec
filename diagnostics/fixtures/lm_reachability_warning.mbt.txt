fn unreachable_catchall(input : String) -> Int {
  lexmatch input {
    re"^.*$" => 1
    _ => 0
  }
}

fn shadowed_case(input : String) -> Int {
  lexmatch input {
    re"^a$" => 1
    re"^a$" => 2
    _ => 0
  }
}

fn empty_language(input : String) -> Int {
  lexmatch input {
    (re"^[]", after=_) => 1
    _ => 0
  }
}
