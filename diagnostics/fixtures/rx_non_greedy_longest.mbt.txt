// RX-017 RX-018
const HIDDEN_NON_GREEDY = re"^a*?"

fn literal(input : String) -> Int {
  lexmatch input with longest {
    re"^a*?" => 1
    re"^a+?" => 2
    re"^a??" => 3
    re"^a{2}?" => 4
    re"^a{2,}?" => 5
    re"^a{2,4}?" => 6
    _ => 0
  }
}

fn referenced(input : String) -> Int {
  lexmatch input with longest {
    HIDDEN_NON_GREEDY => 1
    _ => 0
  }
}

fn lexscan_literal(input : @lexbuf.Lexbuf) -> Int {
  lexscan input with longest {
    re"^a*?" => 1
    re"^a+?" => 2
    re"^a??" => 3
    re"^a{2}?" => 4
    re"^a{2,}?" => 5
    re"^a{2,4}?" => 6
    _ => 0
  }
}

fn lexscan_referenced(input : @lexbuf.Lexbuf) -> Int {
  lexscan input with longest {
    HIDDEN_NON_GREEDY => 1
    _ => 0
  }
}
