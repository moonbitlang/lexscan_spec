// LM-002A
struct UserTarget {}

fn integer_target(input : Int) -> Int {
  lexmatch input { _ => 0 }
}

fn byte_array_target(input : Array[Byte]) -> Int {
  lexmatch input { _ => 0 }
}

fn lexbuf_target(input : @lexbuf.Lexbuf) -> Int {
  lexmatch input { _ => 0 }
}

async fn async_lexbuf_target(input : @lexbuf.AsyncLexbuf) -> Int {
  lexmatch input { _ => 0 }
}

fn scanner_target(input : @lexbuf.StringScanner) -> Int {
  lexmatch input { _ => 0 }
}

fn user_target(input : UserTarget) -> Int {
  lexmatch input { _ => 0 }
}
